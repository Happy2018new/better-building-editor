// __multiversion__
precision highp float;
precision highp int;
// This signals the loading code to prepend either #version 100 or #version 300 es as apropriate.

#include "fragmentVersionCentroid.h"

#if __VERSION__ >= 300
	#ifndef BYPASS_PIXEL_SHADER
		#if defined(TEXEL_AA) && defined(TEXEL_AA_FEATURE)
			_centroid in highp vec2 uv0;
			_centroid in highp vec2 uv1;
		#else
			_centroid in vec2 uv0;
			_centroid in vec2 uv1;
		#endif
	#endif
#else
	#ifndef BYPASS_PIXEL_SHADER
		varying vec2 uv0;
		varying vec2 uv1;
	#endif
#endif

varying vec4 color;

#ifdef FOG
varying vec4 fogColor;
#endif

#include "uniformShaderConstants.h"
#include "util.h"

LAYOUT_BINDING(0) uniform sampler2D TEXTURE_0;
LAYOUT_BINDING(1) uniform sampler2D TEXTURE_1;
LAYOUT_BINDING(2) uniform sampler2D TEXTURE_2;

void main()
{
#ifdef BYPASS_PIXEL_SHADER
	gl_FragColor = vec4(0, 0, 0, 0);
	return;
#else 

#if USE_TEXEL_AA
	vec4 diffuse = texture2D_AA(TEXTURE_0, uv0);
#else
	vec4 diffuse = texture2D(TEXTURE_0, uv0);
#endif
	
#ifdef SEASONS_FAR
	diffuse.a = 1.0;
#endif

#if USE_ALPHA_TEST
	#ifdef ALPHA_TO_COVERAGE
	#define ALPHA_THRESHOLD 0.05
	#else
	#define ALPHA_THRESHOLD 0.5
	#endif
	if(diffuse.a < ALPHA_THRESHOLD)
		discard;
#endif
	
vec4 inColor = color;

#if defined(BLEND)
	diffuse.a *= inColor.a;
#endif

#if !defined(ALWAYS_LIT)
	diffuse *= texture2D( TEXTURE_1, uv1 );
#endif

#ifndef SEASONS
	#if !USE_ALPHA_TEST && !defined(BLEND)
		diffuse.a = inColor.a;
	#endif
	
	diffuse.rgb *= inColor.rgb;
#else
	vec2 uv = inColor.xy;
	diffuse.rgb *= mix(vec3(1.0,1.0,1.0), texture2D( TEXTURE_2, uv).rgb*2.0, inColor.b);
	diffuse.rgb *= inColor.aaa;
	diffuse.a = 1.0;
#endif

#ifdef FOG
	diffuse.rgb = mix( diffuse.rgb, fogColor.rgb, fogColor.a );
#endif

	float shelfTerrainStripe=mod(floor(gl_FragCoord.x/16.0),8.0);
    vec3 shelfTerrainRgb=diffuse.rgb;
    vec3 shelfTerrainLight=vec3(1.0);
#if !defined(ALWAYS_LIT)
    shelfTerrainLight=texture2D(TEXTURE_1,uv1).rgb;
#endif
    vec3 shelfTerrainFlags=vec3(0.0);
#ifdef FANCY
shelfTerrainFlags.r+=1.0/15.0;
#endif
#ifdef FOG
shelfTerrainFlags.r+=2.0/15.0;
#endif
#ifdef ALWAYS_LIT
shelfTerrainFlags.r+=4.0/15.0;
#endif
#ifdef ALPHA_TEST
shelfTerrainFlags.r+=8.0/15.0;
#endif
#ifdef BLEND
shelfTerrainFlags.g+=1.0/15.0;
#endif
#ifdef SEASONS
shelfTerrainFlags.g+=2.0/15.0;
#endif
#ifdef SEASONS_FAR
shelfTerrainFlags.g+=4.0/15.0;
#endif
#ifdef ALLOW_FADE
shelfTerrainFlags.g+=8.0/15.0;
#endif
#ifdef TEXEL_AA
shelfTerrainFlags.b+=1.0/15.0;
#endif
#ifdef ATLAS_TEXTURE
shelfTerrainFlags.b+=2.0/15.0;
#endif
#ifdef TEXEL_AA_FEATURE
shelfTerrainFlags.b+=4.0/15.0;
#endif
#ifdef AS_ENTITY_RENDERER
shelfTerrainFlags.b+=8.0/15.0;
#endif
    if(shelfTerrainStripe<0.5) shelfTerrainRgb=vec3(1.0,1.0,0.0);
    else if(shelfTerrainStripe<1.5) shelfTerrainRgb=color.rgb;
    else if(shelfTerrainStripe<2.5) shelfTerrainRgb=vec3(1.0,0.0,1.0);
    else if(shelfTerrainStripe<3.5) shelfTerrainRgb=shelfTerrainLight;
    else if(shelfTerrainStripe<4.5) shelfTerrainRgb=vec3(0.0,1.0,1.0);
    else if(shelfTerrainStripe<5.5) shelfTerrainRgb=shelfTerrainFlags;
    else if(shelfTerrainStripe<6.5) shelfTerrainRgb=vec3(1.0);
    gl_FragColor=vec4(shelfTerrainRgb,diffuse.a);
	
#endif // BYPASS_PIXEL_SHADER
}
