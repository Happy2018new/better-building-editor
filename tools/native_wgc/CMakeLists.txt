cmake_minimum_required(VERSION 3.20)
project(native_wgc_capture LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "Window capture requires Windows")
endif()
add_executable(native_wgc_capture main.cpp window_capture.cpp)
target_compile_features(native_wgc_capture PRIVATE cxx_std_20)
target_compile_options(native_wgc_capture PRIVATE /utf-8 /EHsc)
set_property(TARGET native_wgc_capture PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(native_wgc_capture PRIVATE d3d11 dxgi dwmapi windowscodecs ole32 runtimeobject user32)

add_executable(native_wgc_burst_capture burst_main.cpp burst_window_capture.cpp)
target_compile_features(native_wgc_burst_capture PRIVATE cxx_std_20)
target_compile_options(native_wgc_burst_capture PRIVATE /utf-8 /EHsc)
set_property(TARGET native_wgc_burst_capture PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(native_wgc_burst_capture PRIVATE d3d11 dxgi dwmapi windowscodecs ole32 runtimeobject user32)
